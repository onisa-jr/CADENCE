\section{Overview}
This supplementary document provides comprehensive architectural parameters, implementation details, reproducible environment specifications, and exhaustive dataset-level evaluation results for all 109 univariate time series datasets in the UCR Time Series Archive across 30 stratified resamples (3,270 total evaluations).

\section{Detailed Hyperparameter Specifications}

\subsection{Branch A: Convolutional Linear Expert}
\begin{itemize}
    \item \textbf{Kernel Generation:} Deterministic MiniRocket transform generating $D_A = 10,000$ features from length-9 kernels with weights in $\{-1, 2\}$.
    \item \textbf{Dilations:} Dyadic log-spaced dilations $d \in \{2^0, 2^1, \dots, \lfloor (L-1)/8 \rfloor\}$.
    \item \textbf{Pooling Operator:} Proportion of Positive Values (PPV): $\phi_{PPV}(X, \omega, b) = \frac{1}{L - 8d} \sum_t \mathbb{I}((X *_{d} \omega)_t > b)$.
    \item \textbf{Normalization:} Channel-wise standard scaling ($\mu=0, \sigma=1$) with numerical stability $\epsilon = 10^{-8}$.
    \item \textbf{Linear Classifier:} RidgeClassifierCV evaluating 15 candidate $\alpha$ values log-spaced in $[10^{-3}, 10^{4}]$ via leave-one-out cross-validation.
    \item \textbf{Probability Calibration:} Softmax over raw linear decision values $f_{A}(X) = W^T \tilde{F}_A + b$.
\end{itemize}

\subsection{Branch B: Distributional Interval Expert}
\begin{itemize}
    \item \textbf{Hydra Grouping:} $G = 16$ competing groups with $K_H = 8$ dilated kernels per group ($16 \times 8 = 128$ features).
    \item \textbf{MomentQuant Cornish-Fisher Expansion:} Evaluated on raw signal ($X$), velocity ($\Delta X_t = X_t - X_{t-1}$), and acceleration ($\Delta^2 X_t = \Delta X_t - \Delta X_{t-1}$) across dyadic intervals up to depth $D = 5$. Each segment produces 9 features (mean, std, skewness, kurtosis, and 5 CF quantiles at $\alpha \in \{0.05, 0.25, 0.50, 0.75, 0.95\}$). Total features: $3 \times 63 \times 9 = 1,701$.
    \item \textbf{FFT Spectral Energy Bands:} 16 log-spaced frequency energy bands, normalized spectral centroid, spectral spread, and 4 cumulative energy quantiles ($\alpha \in \{0.25, 0.50, 0.75, 0.90\}$). Total features: $16 + 2 + 4 = 22$.
    \item \textbf{Decision Head:} Extremely Randomized Trees (ExtraTrees) with $M = 100$ estimators, Shannon entropy splitting criterion, and 10\% random feature subsampling per split (`max_features = 0.1`).
\end{itemize}

\subsection{Confidence-Adaptive Meta-Router}
\begin{itemize}
    \item \textbf{Internal Validation Partition:} Safe 30\% stratified held-out split of $X_{\text{train}}$. Classes with only 1 sample are retained in the training fold.
    \item \textbf{Validation Proxies:} Branch A uses RidgeClassifierCV; Branch B uses a lightweight 50-tree ExtraTrees proxy to minimize validation fitting overhead.
    \item \textbf{Routing Threshold:} $\tau = \pm 0.08$. Margin $\Delta = \text{Val}_A - \text{Val}_B$.
    \item \textbf{Interpolation Rule:} For $|\Delta| \le \tau$, soft-blending weight $w_A = \text{clip}(0.5 + 2.0 \cdot \Delta, 0.15, 0.85)$.
\end{itemize}
